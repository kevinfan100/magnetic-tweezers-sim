! P-numbering: Long 2016 convention (P1=0 deg, etc.)
!
! Simplifications vs Hung CAD:
!   - Protrusions: cylindrical (CYL4), not blocks
!   - Guide_Post: merged into protrusion
!   - Tip: sharp cone (no fillet, straight line to tip)
!
! Changes from Long 2016 marked with [HUNG]
! ================================================================

! ===============================
! ******** PARAMETERS: **********
! ===============================
MESH_SIZE = 5

POLE_NUM = 6
COUNTER  = 0

*dim, POLE_ARRAY,,  POLE_NUM
*dim, PROT_ARRAY,,  POLE_NUM

*afun, DEG
BTOL, 1e-6
/UNITS, MKS

! ---- Hexapole geometric constraints (SAME for all designs) ----
R_norm    = 500e-6
R_norm_xy = R_norm*sqrt(2.0/3.0)
R_norm_z  = R_norm/sqrt(3.0)

! ---- Yoke dimensions [HUNG] ----
YOKE_IN_R  = 80e-3/2.0     ! [HUNG] 40 mm (was 42 mm)
YOKE_OUT_R = 104e-3/2.0    ! [HUNG] 52 mm (was 53 mm)
YOKE_MID_R = (YOKE_IN_R + YOKE_OUT_R)/2.0
YOKE_H     = 2.0e-3

! ---- Protrusion [HUNG] simplified from Pole_Blocks + Guide_Post ----
PROT_R     = 10.0e-3/2.0   ! 5 mm (same as Long 2016)
PROT_H     = 17.0e-3       ! [HUNG] 17 mm (was 7 mm, matches coil height)

! ---- Pole dimensions [HUNG] ----
POLE_R        = 6.35e-3/2       ! [HUNG] 3.175 mm (was 3 mm)
POLE_CONE_LEN = 16e-3           ! [HUNG] 16 mm (was 15 mm, from R/tan(11.3 deg))
! [HUNG] No tip fillet — sharp cone tip (Long 2016 had POLE_TIP_R = 40 um)
! Fillet-related variables removed: POLE_TIP_R, POLE_TIP_CENTER, etc.

! ---- WP center Z offset ----
MT_CTR_ZOFST = -PROT_H - 6.0e-3 + R_norm_z

! ===============================
! ******* YOKE MODELING: ********
! ===============================
WPLANE, 1,0,0,0,,,,,,
WPcsys,-1,0

! **** 1 yoke: ****
CYL4,0,0,YOKE_IN_R,0,YOKE_OUT_R,360,YOKE_H

! **** 6 protrusion: ****
*do, prot_ind, 1, POLE_NUM,1

	WPLANE,1,0,0,0,,,,,,
	WPcsys,-1,0

        *if, prot_ind,LE,3, then
                temp_prot_h = -PROT_H
		temp_prot_ang = 120*(prot_ind - 1)
	*else
		WPOFFS,0,0,YOKE_H
		temp_prot_h = PROT_H
		temp_prot_ang = 120*(prot_ind - 4) + 60
	*endif

	CYL4,YOKE_MID_R*cos(temp_prot_ang),YOKE_MID_R*sin(temp_prot_ang),PROT_R, ,,,temp_prot_h
*enddo

! **** Unite the yoke and protrusion: ****
FLST,2, 1 + POLE_NUM, 6,ORDE, 2
FITEM,2, 1
FITEM,2,-(1+POLE_NUM)
VADD,P51X

*get, vol_yoke_ind, VOLU,,NUM, MAX


! ============================
! **** 3 lower poles: ********
! ============================
X0_LOW = YOKE_MID_R*cos(0)
Y0_LOW = YOKE_MID_R*sin(0)
Z0_LOW = -PROT_H

*do, pole_ind,1,3,1

	WPLANE, 1,0,0,0,,,,,,
	WPcsys,-1,0
        ROTA_ANG = 120*(pole_ind-1)
	WPROTA, ROTA_ANG,

	*get, VL_NUM, VOLU, ,NUM, MAX
	VL_START = VL_NUM + 1
	numstr, VOLU, VL_START

	! **** Lower pole holder: ****
	BLOCK, X0_LOW - 5e-3,  X0_LOW + 11e-3, Y0_LOW - 5e-3,  Y0_LOW + 5e-3,  Z0_LOW - 10e-3, Z0_LOW
	BLOCK, X0_LOW - 5e-3,  X0_LOW - 2e-3,  Y0_LOW - 10e-3, Y0_LOW + 10e-3, Z0_LOW - 10e-3, Z0_LOW - 5e-3
	BLOCK, X0_LOW - 10e-3, X0_LOW - 2e-3,  Y0_LOW - 5e-3,  Y0_LOW + 5e-3,  Z0_LOW - 10e-3, Z0_LOW - 2e-3
	BLOCK, X0_LOW - 10e-3, X0_LOW - 5e-3,  Y0_LOW - 10e-3, Y0_LOW + 10e-3, Z0_LOW - 10e-3, Z0_LOW - 5e-3

	! **** Key points & line [HUNG] sharp tip, no fillet: ****
	! Profile: KP1(tip) -> KP2(cone base) -> KP3(holder edge) -> KP4(holder center)
	! 4 keypoints, 4 straight lines, no LARC
	*get, KP_NUM, KP  , ,NUM, MAX
	*get, LI_NUM, LINE, ,NUM, MAX
	KP_START = KP_NUM + 1
	LI_START = LI_NUM + 1
	numstr, KP,   KP_START
	numstr, LINE, LI_START

	! KP1: tip point (on axis of revolution)
	l1 = R_norm_xy
	l1_ang = atan2( 0 , R_norm_xy )

	! KP2: cone base edge (POLE_R offset, at POLE_CONE_LEN from tip)
	POLE_CONE_LEN_SHIFT = POLE_CONE_LEN + R_norm_xy
	l2 = sqrt( POLE_CONE_LEN_SHIFT**2 + POLE_R**2 )
	l2_ang = atan2( POLE_R , POLE_CONE_LEN )

	! KP3: holder edge (same POLE_R offset)
	l3 = sqrt( (X0_LOW - 10e-3)**2 + POLE_R**2 )
	l3_ang = atan2( POLE_R , (X0_LOW - 10e-3) )

	! KP4: holder center (on axis of revolution)
	l4 = abs( X0_LOW - 10e-3 )
	l4_ang = atan2( 0 , (X0_LOW - 10e-3) )

	K , , l1*cos(ROTA_ANG + l1_ang) , l1*sin(ROTA_ANG + l1_ang), Z0_LOW - 6e-3
	K , , l2*cos(ROTA_ANG + l2_ang) , l2*sin(ROTA_ANG + l2_ang), Z0_LOW - 6e-3
	K , , l3*cos(ROTA_ANG + l3_ang) , l3*sin(ROTA_ANG + l3_ang), Z0_LOW - 6e-3
	K , , l4*cos(ROTA_ANG + l4_ang) , l4*sin(ROTA_ANG + l4_ang), Z0_LOW - 6e-3

	L , KP_START , KP_START + 1,,,,,,,,
	L , KP_START + 1 , KP_START + 2,,,,,,,,
	L , KP_START + 2 , KP_START + 3,,,,,,,,
	L , KP_START + 3 , KP_START,,,,,,,,

	! **** Create Area: ****
	*get, AR_NUM, AREA, ,NUM, MAX
	AR_START = AR_NUM + 1
	numstr, AREA, AR_START

	AL, LI_START, LI_START + 1, LI_START + 2, LI_START + 3

	! **** Rotate area to get volume: ****
	VROTAT, AR_START, ,,,,,KP_START,KP_START + 3,,4

	! **** Unite the volume: ****
	*get, vl_ind_temp, VOLU,,NUM,MAX
	FLST,2,(vl_ind_temp - VL_START + 1),6,ORDE,2
	FITEM,2,VL_START
	FITEM,2,-vl_ind_temp
	VADD,P51X

	*get, vol_pole_ind, VOLU,,NUM,MAX
	VADD, VL_NUM, vol_pole_ind

	*get, vol_unite_ind, VOLU,,NUM,MAX

	BLOCK,0,X0_LOW - 10e-3,Y0_LOW - 10e-3,Y0_LOW + 10e-3,Z0_LOW - 6e-3, Z0_LOW

	VSBV, vol_unite_ind,vol_unite_ind + 1

	GPLOT

*enddo


! ===========================
! **** 3 upper poles: *******
! ===========================
	X0_HIGH = YOKE_MID_R*cos(0)
	Y0_HIGH = YOKE_MID_R*sin(0)
	Z0_HIGH = YOKE_H + PROT_H

*do, pole_ind,4,6,1
	WPLANE, 1,0,0,0,,,,,,
	WPcsys,-1,0
        ROTA_ANG = 120*(pole_ind-4) + 60
	WPROTA, ROTA_ANG,

	*get, VL_NUM, VOLU, ,NUM, MAX
	VL_START = VL_NUM + 1
	numstr, VOLU, VL_START

	! **** Upper pole holder: ****
	BLOCK, X0_HIGH - 3.5e-3,  X0_HIGH + 8.5e-3, Y0_HIGH - 5e-3,  Y0_HIGH + 5e-3,  Z0_HIGH, Z0_HIGH + 10e-3
	BLOCK, X0_HIGH - 6.5e-3,  X0_HIGH - 3.5e-3, Y0_HIGH - 11e-3, Y0_HIGH + 11e-3, Z0_HIGH, Z0_HIGH + 10e-3
	BLOCK, X0_HIGH - 16.5e-3, X0_HIGH - 6.5e-3, Y0_HIGH - 11e-3, Y0_HIGH + 11e-3, Z0_HIGH, Z0_HIGH + 10e-3

        tip_upper_x = R_norm_xy*cos(ROTA_ANG)
        tip_upper_y = R_norm_xy*sin(ROTA_ANG)
	tip_upper_z = Z0_LOW - 6e-3 + R_norm_z*2

	end_upper_x0 = X0_HIGH - 11.5e-3
	end_upper_y0 = Y0_HIGH
        end_upper_z0 = Z0_HIGH + 5e-3
	end_upper_x = sqrt( end_upper_x0**2 + end_upper_y0**2 )*cos(ROTA_ANG)
	end_upper_y = sqrt( end_upper_x0**2 + end_upper_y0**2 )*sin(ROTA_ANG)
	end_upper_z = end_upper_z0

	INCLINE_VECTORx = end_upper_x - tip_upper_x
	INCLINE_VECTORy = end_upper_y - tip_upper_y
	INCLINE_VECTORz = end_upper_z - tip_upper_z

	INCLINE_ANG = atan2( INCLINE_VECTORz, sqrt(INCLINE_VECTORx**2+INCLINE_VECTORy**2) )

	! **** Key points & line [HUNG] sharp tip, no fillet: ****
	! Profile: KP1(tip) -> KP2(cone base) -> KP3(holder edge) -> KP4(holder center)
	*get, KP_NUM, KP  , ,NUM, MAX
	*get, LI_NUM, LINE, ,NUM, MAX
	KP_START = KP_NUM + 1
	LI_START = LI_NUM + 1
	numstr, KP,   KP_START
	numstr, LINE, LI_START

	! KP1: tip point
        p1x = tip_upper_x
	p1y = tip_upper_y
	p1z = tip_upper_z

	! KP2: cone base edge (POLE_R offset from axis, at POLE_CONE_LEN from tip)
	POLE_CONE_LEN_SHIFT = POLE_CONE_LEN + R_norm_xy
	p2x = p1x + POLE_CONE_LEN_SHIFT*cos(INCLINE_ANG)*cos(ROTA_ANG) - POLE_R*sin(ROTA_ANG)
	p2y = p1y + POLE_CONE_LEN_SHIFT*cos(INCLINE_ANG)*sin(ROTA_ANG) + POLE_R*cos(ROTA_ANG)
	p2z = p1z + POLE_CONE_LEN_SHIFT*sin(INCLINE_ANG)

	! KP3: holder edge
	p3x = end_upper_x - POLE_R*sin(ROTA_ANG)
	p3y = end_upper_y + POLE_R*cos(ROTA_ANG)
	p3z = end_upper_z

	! KP4: holder center (on axis of revolution)
	p4x = end_upper_x
	p4y = end_upper_y
	p4z = end_upper_z

	K,, p1x , p1y , p1z
	K,, p2x , p2y , p2z
	K,, p3x , p3y , p3z
	K,, p4x , p4y , p4z

	L , KP_START , KP_START + 1,,,,,,,,
	L , KP_START + 1 , KP_START + 2,,,,,,,,
	L , KP_START + 2 , KP_START + 3,,,,,,,,
	L , KP_START + 3 , KP_START,,,,,,,,

	! **** Create Area: ****
	*get, AR_NUM, AREA, ,NUM, MAX
	AR_START = AR_NUM + 1
	numstr, AREA, AR_START

	AL, LI_START, LI_START + 1, LI_START + 2, LI_START + 3

	! **** Rotate area to get volume: ****
	VROTAT, AR_START, ,,,,,KP_START,KP_START + 3,,4

	! **** Unite the volume: ****
	*get, vl_ind_temp, VOLU,,NUM,MAX
	FLST,2,(vl_ind_temp - VL_START + 1),6,ORDE,2
	FITEM,2,VL_START
	FITEM,2,-vl_ind_temp
	VADD,P51X

	*get, vol_pole_ind, VOLU,,NUM,MAX
	VADD, VL_NUM, vol_pole_ind

	GPLOT

*enddo


NUMCMP,VOLU
V_ID_MT = 1


! ==========================
! **** Create Sphere: ******
! ==========================
SPH_FINE_R = 7e-3
SPH_OFST = Z0_LOW - 6e-3 + R_norm_z
AIR_CYL_R = 80e-3
AIR_CYL_H = 70e-3

WPLANE, 1,0,0,0,,,,,,
WPcsys,-1,0
WPOFFS,0,0,SPH_OFST

SPHERE, 0, SPH_FINE_R, 0,360

WPOFFS,0,0, -SPH_OFST + YOKE_H/2 - AIR_CYL_H/2
CYL4,0,0,AIR_CYL_R, ,,,AIR_CYL_H


V_ID_AIR_SPH = 2
V_ID_AIR_CYL = 3


! ===================
! **** OverLap: *****
! ===================
VOVLAP,ALL
NUMCMP,VOLU

! [NOTE] Volume IDs after overlap may differ from Long 2016.
! Verify with VPLOT before proceeding.
! Expected: 6 pole volumes + yoke + air sphere + air cylinder = 9

VGLUE,ALL

VPLOT


! =============================
! **** Magnetic analysis: *****
! =============================

! **** Material definition: ****
MAT_AIR1 = 1
MAT_MT   = 2
MAT_AIR2 = 3
mp , murx , MAT_AIR1 , 1
mp , murx , MAT_MT   , 280   ! [HUNG] same as Long 2016
mp , murx , MAT_AIR2 , 1

! **** Element definition: ****
ET_nocurr = 1
ET_curr   = 2
ET , ET_nocurr , SOLID96
ET , ET_curr   , SOURC36


! **** Define current sources [HUNG] updated coil dimensions: ****
COIL_OUT_R = 12e-3      ! [HUNG] 12 mm (was 8 mm)
COIL_IN_R  = 10e-3      ! [HUNG] 10 mm (was 5 mm)
*dim, CURR_ARRAY,,POLE_NUM
CURR_ARRAY(1) = 1        ! Only Coil 1 excited
CURR_ARRAY(2) = 0
CURR_ARRAY(3) = 0
CURR_ARRAY(4) = 0
CURR_ARRAY(5) = 0
CURR_ARRAY(6) = 0
TURNS = 70               ! Same as Long 2016
COIL_THK = COIL_OUT_R - COIL_IN_R
COIL_R = (COIL_IN_R + COIL_OUT_R)/2
COIL_H = PROT_H          ! [HUNG] 17 mm (was 7 mm)
R,1,1,TURNS*CURR_ARRAY(1),COIL_THK,COIL_H
R,2,1,TURNS*CURR_ARRAY(2),COIL_THK,COIL_H
R,3,1,TURNS*CURR_ARRAY(3),COIL_THK,COIL_H
R,4,1,TURNS*CURR_ARRAY(4),COIL_THK,COIL_H
R,5,1,TURNS*CURR_ARRAY(5),COIL_THK,COIL_H
R,6,1,TURNS*CURR_ARRAY(6),COIL_THK,COIL_H

WPLANE, 1,0,0,0,,,,,,
WPcsys,-1,0

NUMCMP,NODE

*get, NODE_NUM,NODE,,NUM,MAX
NODE_START = NODE_NUM + 1

! **** Lower coils: ****
*do, pole_ind,1,3,1

	ROTA_ANG = 120*(pole_ind-1)
	X0_LOW = YOKE_MID_R*cos(ROTA_ANG)
	Y0_LOW = YOKE_MID_R*sin(ROTA_ANG)
	Z0_LOW = -PROT_H

        node1 = NODE_START + 3*(pole_ind-1)
	node2 = NODE_START + 3*(pole_ind-1) + 1
	node3 = NODE_START + 3*(pole_ind-1) + 2
	N , node1 , X0_LOW + COIL_R , Y0_LOW          , Z0_LOW + COIL_H/2
	N , node2 , X0_LOW          , Y0_LOW + COIL_R , Z0_LOW + COIL_H/2
	N , node3 , X0_LOW          , Y0_LOW          , Z0_LOW + COIL_H/2
	TYPE,ET_curr
	REAL,pole_ind
	E,node1,node2,node3
	/ESHAPE,1
*enddo

! **** Upper coils: ****
*do, pole_ind,4,6,1

	ROTA_ANG = 120*(pole_ind-4) + 60
	X0_HIGH = YOKE_MID_R*cos(ROTA_ANG)
	Y0_HIGH = YOKE_MID_R*sin(ROTA_ANG)
	Z0_HIGH = YOKE_H + PROT_H

        node1 = NODE_START + 3*(pole_ind-1)
	node2 = NODE_START + 3*(pole_ind-1) + 1
	node3 = NODE_START + 3*(pole_ind-1) + 2
	N , node1 , X0_HIGH + COIL_R , Y0_HIGH          , Z0_HIGH - COIL_H/2
	N , node2 , X0_HIGH          , Y0_HIGH + COIL_R , Z0_HIGH - COIL_H/2
	N , node3 , X0_HIGH          , Y0_HIGH          , Z0_HIGH - COIL_H/2
	TYPE,ET_curr
	REAL,pole_ind
	E,node1,node2,node3
	/ESHAPE,1
*enddo


! **** Mesh: ****
VSEL,s,VOLU,,1,6
VSEL,a,VOLU,,8

VSEL,s,VOLU,,7

! ================================================
