! ====================================================================
! [ADDED] Kuo Quadrupole simulation - SINGLE COIL excitation
!         For computing per-pole self-inductance to compare PRSTAB 32 nH/pole
!
! Differences from MT_Kuo_Quadrupole_Sim_Dipole.txt:
!   1. T_Z = 0.055 mm (PRSTAB prototype thickness, not JMEMS 0.100)
!   2. CURR_ARRAY = [0, 0, +1, 0]: only Pole 3 energized (NI = +4 A·turns)
!
! Pole length L_P = 0.46 mm (fitted via JMEMS T=100 µm sweep, transferable to
! T=55 µm because pole length is in-plane design parameter independent of z).
!
! Output: sim_solved.db; run material-split diagnostic to extract L_self per pole.
! ====================================================================

/CLEAR, NOSTART
/PREP7
*afun, DEG
BTOL, 1e-7

! ========== Units: MKS ==========
MM = 1.0e-3

! ========== Read sweep variable L_P from external param file ==========
/INPUT, 'lp_param', 'inp', 'G:\my_workspace\code\magnetic-tweezers-sim\kuo\apdl\sim\'

! ========== Geometry parameters ==========
HALF_TAPER     = 7.5
TIP_FILLET_R   = 0.045*MM
T_Z            = 0.055*MM       ! [MODIFIED] PRSTAB prototype thickness (was 0.100 in dipole sim)
YOKE_WALL      = 0.200*MM

FILLET_CENTER_R = TIP_FILLET_R/sin(HALF_TAPER)
TANGENT_R       = FILLET_CENTER_R*cos(HALF_TAPER)
MAGNET_R        = FILLET_CENTER_R - TIP_FILLET_R

YOKE_RI         = MAGNET_R + L_P_VAL
YOKE_RO         = YOKE_RI + YOKE_WALL

Z_BOT = -T_Z/2

AIR_R = 5.0*MM
AIR_Z = 10.0*MM

! ========== Coil parameters ==========
TURNS         = 4
DRIVE_CURRENT = 1.0
COIL_R        = (YOKE_RI + MAGNET_R)/2
COIL_MEAN_R   = 0.10*MM
COIL_DY       = 0.025*MM
COIL_DZ       = 0.20*MM

! ========== Single-coil excitation: only Pole 3 energized ==========
*dim, CURR_ARRAY, , 4
CURR_ARRAY(1) = 0    ! OFF
CURR_ARRAY(2) = 0    ! OFF
CURR_ARRAY(3) = 1    ! Pole 3 only, NI = +4 A·turns
CURR_ARRAY(4) = 0    ! OFF

! ========== Build magnet ==========
K, 1, 0, 0, Z_BOT

*do, I_POLE, 1, 4
    ANG_C = -135 + 90*(I_POLE-1)
    ANG_S = ANG_C - HALF_TAPER
    ANG_N = ANG_C + HALF_TAPER

    *get, KP_NUM, KP, , NUM, MAX
    KP_S = KP_NUM + 1

    K, KP_S,   YOKE_RI*cos(ANG_S),         YOKE_RI*sin(ANG_S),         Z_BOT
    K, KP_S+1, YOKE_RI*cos(ANG_N),         YOKE_RI*sin(ANG_N),         Z_BOT
    K, KP_S+2, TANGENT_R*cos(ANG_N),       TANGENT_R*sin(ANG_N),       Z_BOT
    K, KP_S+3, TANGENT_R*cos(ANG_S),       TANGENT_R*sin(ANG_S),       Z_BOT
    K, KP_S+4, FILLET_CENTER_R*cos(ANG_C), FILLET_CENTER_R*sin(ANG_C), Z_BOT
    K, KP_S+5, MAGNET_R*cos(ANG_C),        MAGNET_R*sin(ANG_C),        Z_BOT

    *get, LI_NUM, LINE, , NUM, MAX
    LI_S = LI_NUM + 1
    NUMSTR, LINE, LI_S

    LARC, KP_S,   KP_S+1, 1, YOKE_RI
    L,    KP_S+1, KP_S+2
    BSPLIN, KP_S+2, KP_S+5, KP_S+3
    L,    KP_S+3, KP_S

    *get, AR_NUM, AREA, , NUM, MAX
    AR_S = AR_NUM + 1
    NUMSTR, AREA, AR_S
    AL, LI_S, LI_S+1, LI_S+2, LI_S+3
*enddo

WPOFFS, 0, 0, Z_BOT
PCIRC, YOKE_RO, YOKE_RI, 0, 360
WPOFFS, 0, 0, -Z_BOT

ASEL, ALL
VEXT, ALL, , , 0, 0, T_Z
VSEL, ALL
VADD, ALL
*get, V_MAG, VOLU, , NUM, MAX

! ========== Air domain ==========
CYLIND, AIR_R, 0, -AIR_Z, AIR_Z, 0, 360
ALLSEL, ALL
VSEL, ALL
VSEL, U, VOLU, , V_MAG
*get, V_AIR_CYL, VOLU, , NUM, MAX
ALLSEL, ALL

VSBV, V_AIR_CYL, V_MAG, , DELETE, KEEP

ALLSEL, ALL
VSEL, ALL
VSEL, U, VOLU, , V_MAG
*get, V_AIR, VOLU, , NUM, MAX
ALLSEL, ALL

! ========== Element types ==========
ET, 1, SOLID96
ET, 2, SOURC36

MP, MURX, 1, 8000
MP, MURX, 2, 1

VSEL, S, VOLU, , V_MAG
VATT, 1, , 1
VSEL, S, VOLU, , V_AIR
VATT, 2, , 1
ALLSEL, ALL

! ========== Mesh ==========
MSHAPE, 1, 3D
MSHKEY, 0

VSEL, S, VOLU, , V_MAG
ESIZE, 0.05*MM
VMESH, ALL
ALLSEL, ALL

VSEL, S, VOLU, , V_AIR
ESIZE, 0.7*MM
VMESH, ALL
ALLSEL, ALL

*get, NN, NODE, 0, COUNT
*get, NE, ELEM, 0, COUNT
*msg, INFO, NN, NE
 Mesh complete: nodes=%I  elements=%I

! ========== SOURC36 coils ==========
*do, sc_ri, 1, 4
    NI_VALUE = TURNS*CURR_ARRAY(sc_ri)*DRIVE_CURRENT
    R, sc_ri, 1, NI_VALUE, COIL_DY, COIL_DZ
*enddo

NUMCMP, NODE
*get, NODE_NUM, NODE, , NUM, MAX
NODE_START = NODE_NUM + 1

*do, sc_pi, 1, 4
    sc_ang = -135 + 90*(sc_pi-1)
    sc_cx = COIL_R*cos(sc_ang)
    sc_cy = COIL_R*sin(sc_ang)
    sc_cz = 0

    sc_n1 = NODE_START + (sc_pi-1)*3
    sc_n2 = sc_n1 + 1
    sc_n3 = sc_n1 + 2

    N, sc_n1, sc_cx - COIL_MEAN_R*sin(sc_ang), sc_cy + COIL_MEAN_R*cos(sc_ang), sc_cz
    N, sc_n2, sc_cx, sc_cy, sc_cz + COIL_MEAN_R
    N, sc_n3, sc_cx, sc_cy, sc_cz

    TYPE, 2
    REAL, sc_pi
    E, sc_n1, sc_n2, sc_n3
*enddo

*msg, INFO
 SOURC36 coils built (single-coil mode: only Pole 3 active)

ALLSEL, ALL
SAVE, 'sim_built', 'db'

! ========== /SOLU ==========
FINISH
/SOLU

ALLSEL, ALL
NSEL, S, LOC, Z, AIR_Z - 1.0e-6, AIR_Z + 1.0e-6
D, ALL, MAG, 0
ALLSEL, ALL
NSEL, S, LOC, Z, -AIR_Z - 1.0e-6, -AIR_Z + 1.0e-6
D, ALL, MAG, 0
ALLSEL, ALL
LOCAL, 99, 1, 0, 0, 0
CSYS, 99
NSEL, S, LOC, X, AIR_R*0.99, AIR_R*1.01
D, ALL, MAG, 0
CSYS, 0
ALLSEL, ALL

magsolv, 3, , , , , 1

FINISH

/POST1
SET, LAST
SAVE, 'sim_solved', 'db'
FINISH

/EXIT, NOSAV
